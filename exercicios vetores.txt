Exercício 5 - Difícil:


#include <stdio.h>

int main() {
    int vetor[10], unicos[10];
    int qtdUnicos = 0;
    int repetido;

    printf("Digite 10 inteiros:\n");

    for (int i = 0; i < 10; i++) {
        scanf("%d", &vetor[i]);
    }

    
    for (int i = 0; i < 10; i++) {
        repetido = 0;

        for (int j = 0; j < qtdUnicos; j++) {
            if (vetor[i] == unicos[j]) {
                repetido = 1;
                break;
            }
        }

        if (!repetido) {
            unicos[qtdUnicos] = vetor[i];
            qtdUnicos++;
        }
    }
    printf("\nPrimeiro vetor: ");
    for (int i = 0; i < 10; i++) {
        printf("%d ", vetor[i]);
    }
    
    printf("\nVetor sem repeticoes: ");
    for (int i = 0; i < qtdUnicos; i++) {
        printf("%d ", unicos[i]);
    }

    printf("\n");

    return 0;
}

int main() {

            maior = num;
        }else if(num < menor){
            menor = num;
        }
        soma = soma + num;
    }

    for(i = 0; i < 10; i++){

    printf("\nMenor:\t%.1f", menor);
    printf("\nSoma :\t%.1f", soma);

}

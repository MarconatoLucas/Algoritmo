#include <stdio.h>
#define N 3

int main(){
    typedef struct{
        char nome[50];
        int matricula;
        float nota;
    }Aluno;
    
    Aluno turma[N];
    int i;

    for(i = 0; i < N; i++){
        printf("Aluno %d\n", i + 1);
        scanf("%49s %d %f", &turma[i].nome, &turma[i].matricula, &turma[i].nota);
    }

    int idx = 0;
    for(i = 1; i < N; i++){
        if(turma[i].nota > turma[idx].nota){
            idx = 1;
        }
    }

    printf("Melhor: %s %.1f\n", turma[idx].nome, turma[idx].nota);

    
}
